API Documentation

POST /api/order

Place a topup order via API. Requires API key in header.

Headers:
Authorization: api_sk_your_key

Body (JSON):

{
    "service": "ff",
    "uid": "123456789",
    "product": "Weekly"
}

For Mobile Legends / Magic Chess add server_id:

{
    "service": "mg",
    "uid": "7154836339",
    "server_id": "11345",
    "product": "Weekly"
}

Success Response:

{
    "status": "success",
    "message": "Order placed successfully",
    "order_id": 42,
    "command": "/ff 123456789 Weekly"
}

import requests
url = "https://yourdomain.com/api/order"
headers = {"Authorization": "api_sk_your_key"}
data = {"service":"ff","uid":"123456789","product":"Weekly"}
r = requests.post(url, json=data, headers=headers)
print(r.json())
fetch("https://yourdomain.com/api/order", {
    method:"POST",
    headers:{"Authorization":"api_sk_your_key","Content-Type":"application/json"},
    body:JSON.stringify({service:"ff",uid:"123456789",product:"Weekly"})
}).then(r=>r.json()).then(console.log);
<?php
$ch = curl_init("https://yourdomain.com/api/order");
curl_setopt($ch, CURLOPT_POST, 1);
curl_setopt($ch, CURLOPT_HTTPHEADER, ["Authorization: api_sk_your_key", "Content-Type: application/json"]);
curl_setopt($ch, CURLOPT_POSTFIELDS, json_encode(["service"=>"ff","uid"=>"123456789","product"=>"Weekly"]));
curl_setopt($ch, CURLOPT_RETURNTRANSFER, true);
echo curl_exec($ch);
curl_close($ch);
curl -X POST https://yourdomain.com/api/order \
-H "Authorization: api_sk_your_key" \
-H "Content-Type: application/json" \
-d '{"service":"ff","uid":"123456789","product":"Weekly"}'
